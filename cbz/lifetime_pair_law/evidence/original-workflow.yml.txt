name: CBZ conditional pair-law countermodel

jobs:

    runs-on: ubuntu-latest
    timeout-minutes: 45
    env:
      LEAN_NUM_THREADS: '2'
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
        with:

      - uses: leanprover/lean-action@24581d74d674803a25a81b10b059c5c0cdd344c6

          lake-package-directory: specifications/lean_statements_20261010
          auto-config: 'false'
          build: 'false'
          test: 'false'
          lint: 'false'
          use-mathlib-cache: 'true'

          set -euo pipefail
          cd specifications/lean_statements_20261010
          git rev-parse HEAD

        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9
