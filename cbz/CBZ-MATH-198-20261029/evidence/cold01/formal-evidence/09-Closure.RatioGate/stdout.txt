def UniversalLaw.Spec.P15.rhoTest : ℕ → ℕ → ℝ → ℝ :=
fun a d pBound => pBound ^ (a * d - a) / (1 - pBound) ^ (a * d + 1)
UniversalLaw.CertifiedFrontier.ratio_exponents (a d : ℕ) (ha : 1 ≤ a) (hd : 2 ≤ d) :
  1 ≤ a * d - a ∧ a * d + 1 ≤ 3 * (a * d - a)
UniversalLaw.CertifiedFrontier.rhoTest_factor (a d : ℕ) (p : ℝ) (ha : 1 ≤ a) (hd : 2 ≤ d) (hp : p < 1) :
  UniversalLaw.Spec.P15.rhoTest a d p = (p / (1 - p) ^ 3) ^ (a * d - a) * (1 - p) ^ (3 * (a * d - a) - (a * d + 1))
UniversalLaw.CertifiedFrontier.rhoTest_envelope (a d : ℕ) (p : ℝ) (ha : 1 ≤ a) (hd : 2 ≤ d) (hp0 : 0 ≤ p) (hp1 : p < 1)
  (hgate : p ≤ (1 - p) ^ 3) : UniversalLaw.Spec.P15.rhoTest a d p ≤ p / (1 - p) ^ 3
UniversalLaw.CertifiedFrontier.rhoTest_uniform_iff (p : ℝ) (hp0 : 0 ≤ p) (hp1 : p < 1) :
  (∀ (a d : ℕ), 1 ≤ a → 2 ≤ d → UniversalLaw.Spec.P15.rhoTest a d p ≤ 1) ↔ p ≤ (1 - p) ^ 3
UniversalLaw.CertifiedFrontier.rhoTest_zero (a d : ℕ) (ha : 1 ≤ a) (hd : 2 ≤ d) :
  UniversalLaw.Spec.P15.rhoTest a d 0 = 0
UniversalLaw.CertifiedFrontier.rhoTest_one_totalized (a d : ℕ) : UniversalLaw.Spec.P15.rhoTest a d 1 = 0
UniversalLaw.CertifiedFrontier.rhoTest_one_excluded :
  (∀ (a d : ℕ), 1 ≤ a → 2 ≤ d → UniversalLaw.Spec.P15.rhoTest a d 1 ≤ 1) ∧ ¬1 ≤ (1 - 1) ^ 3
