import ResearchFormalCoreR1.GaussianScalarMoments

open MeasureTheory ProbabilityTheory ResearchFormalCoreR1
open scoped NNReal

#check gm8C8
#check gm8_standard_integrable
#check gm8_standard_nonneg
#check gm8_scalar_integrable_bound

example (m B V : ℝ) (v : ℝ≥0) (hB : 0 ≤ B) (hV : 0 ≤ V)
    (hm : |m| ≤ B) (hv : (v : ℝ) ≤ V) :
    Integrable (fun x : ℝ => |x| ^ 8) (gaussianReal m v) ∧
      (∫ x : ℝ, |x| ^ 8 ∂gaussianReal m v) ≤
        128 * (B ^ 8 + V ^ 4 * gm8C8) :=
  gm8_scalar_integrable_bound m B V v hB hV hm hv

example : Integrable (fun x : ℝ => |x| ^ 8) (gaussianReal 1 0) ∧
    (∫ x : ℝ, |x| ^ 8 ∂gaussianReal 1 0) ≤ 128 := by
  simpa using gm8_scalar_integrable_bound 1 1 0 0
    (by norm_num) (by norm_num) (by norm_num) (by norm_num)
