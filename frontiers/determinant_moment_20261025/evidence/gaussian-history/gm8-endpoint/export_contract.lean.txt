example {Ω : Type*} [MeasurableSpace Ω] (μ : Measure Ω) [IsProbabilityMeasure μ]
    (J : Ω → Fin 6 → ℝ) (hJ : Measurable J) (m : Fin 6 → ℝ) (v : Fin 6 → ℝ≥0)
    (B V r : ℝ) (hB : 0 ≤ B) (hV : 0 ≤ V) (hr : 0 < r) (hr1 : r ≤ 1)
    (hmean : ∀ i, |m i| ≤ B) (hvar : ∀ i, (v i : ℝ) ≤ V)
    (hlaw : ∀ i, HasLaw (fun x => J x i) (gaussianReal (m i) (v i)) μ) :
    Integrable (fun x => egm_weight r (J x)^2) μ ∧
      (∫ x, egm_weight r (J x)^2 ∂μ) ≤ 16 * gm8FiniteM8 6 B V * r^4 :=
  egm_gaussian_moment μ J hJ m v B V r hB hV hr hr1 hmean hvar hlaw
