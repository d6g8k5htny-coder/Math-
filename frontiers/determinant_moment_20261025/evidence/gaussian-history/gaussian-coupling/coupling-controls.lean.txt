import ResearchFormalCoreR1.GaussianCoupling
namespace ResearchFormalCoreR1
open MeasureTheory ProbabilityTheory Filter Matrix WithLp
open scoped Topology MatrixOrder Matrix.Norms.L2Operator

-- Nonzero changing mean and rank drop to zero covariance, on the full real filter.
example (z : EuclideanSpace ℝ (Fin 1)) :
    Tendsto (fun r : ℝ => gc_coupledVector (toLp 2 (fun _ : Fin 1 => 3+r))
      (Matrix.diagonal (fun _ : Fin 1 => r^2)) z) (𝓝 0)
      (𝓝 (gc_coupledVector (toLp 2 (fun _ : Fin 1 => 3)) 0 z)) := by
  apply gc_coupledVector_tendsto (𝓝 (0 : ℝ))
      (fun r : ℝ => toLp 2 (fun _ : Fin 1 => 3+r))
      (fun r : ℝ => Matrix.diagonal (fun _ : Fin 1 => r^2))
      (toLp 2 (fun _ : Fin 1 => 3)) 0
  · have hc : Continuous (fun r : ℝ => toLp 2 (fun _ : Fin 1 => 3+r)) := by fun_prop
    simpa using hc.tendsto 0
  · have hc : Continuous (fun r : ℝ => Matrix.diagonal (fun _ : Fin 1 => r^2)) := by fun_prop
    simpa using hc.tendsto 0
  · filter_upwards [] with r
    exact (Matrix.posSemidef_diagonal_iff).2 (fun _ => sq_nonneg r)
  · exact Matrix.PosSemidef.zero

-- Its literal singular law is still the same declared multivariate Gaussian.
example : HasLaw (gc_coupledVector (toLp 2 (fun _ : Fin 1 => (3 : ℝ))) 0)
    (multivariateGaussian (toLp 2 (fun _ : Fin 1 => (3 : ℝ))) 0)
    (stdGaussian (EuclideanSpace ℝ (Fin 1))) :=
  gc_coupledVector_hasLaw _ _

#check gc_coupledVector
#check gc_coupledVector_hasLaw
#check gc_coupledVector_tendsto
#print axioms gc_coupledVector_hasLaw
#print axioms gc_coupledVector_tendsto
end ResearchFormalCoreR1
