example : (Measure.dirac (toLp 2 (fun _ : Fin 1 => (3 : ℝ)))) =
    multivariateGaussian (toLp 2 (fun _ : Fin 1 => (3 : ℝ))) (0 : Matrix (Fin 1) (Fin 1) ℝ) := by
  have hZ : HasGaussianLaw (fun _ : Unit => (fun _ : Fin 1 => (3 : ℝ))) (Measure.dirac ()) := by
    refine ⟨measurable_const.aemeasurable, ?_⟩
    rw [Measure.map_dirac' measurable_const]
    infer_instance
  have h := finiteGaussian_law_parameters hZ (fun _ => 3) (by simp)
    (0 : Matrix (Fin 1) (Fin 1) ℝ) (by intro i j; simp)
  simpa using h.2
