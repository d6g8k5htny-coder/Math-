example {L : ℝ} (hL : L ≠ 0) : sourceContactCoefficient L 5 (sourceContactMode 5) ≠
    sourceContactAmplitude L 5 * (-2 * sourceC L ^ 3) := by
  have hcol : sourceContactCoefficient L 5 (sourceContactMode 5) =
      sourceContactAmplitude L 5 * (-8 * sourceC L ^ 3) := by
    simpa [contactWitness] using sourceContact_selected_columns L 5 5
  rw [hcol]
  intro h
  have hc : sourceC L ≠ 0 := div_ne_zero
    (mul_ne_zero (by norm_num) (ne_of_gt Real.pi_pos)) hL
  have hp : sourceContactAmplitude L 5 * sourceC L ^ 3 ≠ 0 :=
    mul_ne_zero (sourceContactAmplitude_pos hL 5).ne' (pow_ne_zero _ hc)
  apply hp
  nlinarith [h]
