example {L : ℝ} (hL : L ≠ 0) :
    regularizedPinCoefficient L 0 5 (sourceContactMode 5) ≠ 0 := by
  rw [regularizedPinCoefficient_zero]
  change sourceContactCoefficient L 5 (sourceContactMode 5) ≠ 0
  have hcol : sourceContactCoefficient L 5 (sourceContactMode 5) =
      sourceContactAmplitude L 5 * (-8 * sourceC L ^ 3) := by
    simpa [I4ContactGram.contactWitness] using sourceContact_selected_columns L 5 5
  rw [hcol]
  have hc : sourceC L ≠ 0 := div_ne_zero
    (mul_ne_zero (by norm_num) (ne_of_gt Real.pi_pos)) hL
  exact mul_ne_zero (sourceContactAmplitude_pos hL 5).ne'
    (mul_ne_zero (by norm_num) (pow_ne_zero _ hc))
