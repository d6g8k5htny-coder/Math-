example {L : ℝ} (hL : L ≠ 0) (r b : ℝ) (hr : r ≠ 0)
    (hG : IsUnit (sourcePinCovarianceAt L r))
    [IsProbabilityMeasure (sourceSmoothWeightedLaw hL r b)]
    (hZ : 0 < ∫ ω, sourceSmoothWeight L r ω ∂sourceSmoothLaw hL r b) :
    sourceSmoothWeightedLaw hL r b
      {ω | rawPins L r ω = ![b,b-r^3/6,0,0,0,0] ∧ sourceSmoothTyped L r ω} = 1 := by
  apply (mem_ae_iff_prob_eq_one
    ((measurableSet_eq_fun (sourceRawPins_measurable hL r hr) measurable_const).inter
      (sourceSmoothTyped_measurableSet hL r hr))).1
  have hraw : ∀ᵐ ω ∂sourceSmoothWeightedLaw hL r b,
      rawPins L r ω = ![b,b-r^3/6,0,0,0,0] := by
    apply (mem_ae_iff_prob_eq_one
      (measurableSet_eq_fun (sourceRawPins_measurable hL r hr) measurable_const)).2
    exact sourceSmoothWeightedLaw_raw_pins hL r b hr hG
  have htype : ∀ᵐ ω ∂sourceSmoothWeightedLaw hL r b, sourceSmoothTyped L r ω := by
    apply (mem_ae_iff_prob_eq_one (sourceSmoothTyped_measurableSet hL r hr)).2
    exact sourceSmoothWeightedLaw_typed hL r b hr hZ
  exact hraw.and htype
